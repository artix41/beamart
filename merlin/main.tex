\newcommand{\overviewSentence}{
  \catchterm<2>{Most magic-state preparation protocols} can be \catchterm<3>{simulated} in \catchterm<4>{polynomial-time}
}

% One Beamer frame produces four successive pages in the PDF.
\begin{frame}{Overview}
  \overviewstatement{\overviewSentence}

  \only<2>{%
    \termexplanation{1}{Most magic-state preparation protocols}{%
      Gates outside the Clifford group, such as $T$ and CCZ.\par
      These resources occur in many fault-tolerant protocols.%
    }%
  }
  \only<3>{%
    \termexplanation{2}{Simulated}{%
      The process of running the simulation on a classical computer.\par
      This allows for the study of quantum systems without the need for actual quantum hardware.%
    }%
  }
  \only<4>{%
    \termexplanation{3}{Polynomial time}{%
      For the supported circuit class, the simulation cost grows
      polynomially with circuit size.\par
      The claim is restricted to this class of circuits.%
    }%
  }
\end{frame}

    \emphword{Initial state}: $\ket{0}^{\otimes n} \ket{+}^{\otimes m}$

    \emphword{Question}: what happens when we apply $G_3$ gates to $\ket{0}^{\otimes n} \ket{+}^{\otimes m}$
